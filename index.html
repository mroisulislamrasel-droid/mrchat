const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server, { cors: { origin: "*" } }); // রুম ডেটা ও ৮টি মাইক সিটের স্টেট let rooms = { "349200": { title: "জিরো পয়েন্ট ফ্যামিলি", seats: Array(8).fill(null), // ৮টি সিট খালি messages: [] } }; io.on('connection', (socket) => { console.log('User connected:', socket.id); // ভয়েস রুমে জয়েন করা socket.on('join_room', (roomId) => { socket.join(roomId); if (rooms[roomId]) { socket.emit('room_data', rooms[roomId]); } }); // মাইক সিট ধরা বা ছেড়ে দেওয়া (8-Seat Mic System) socket.on('toggle_seat', ({ roomId, seatIndex, user }) => { if (rooms[roomId]) { let currentSeat = rooms[roomId].seats[seatIndex]; rooms[roomId].seats[seatIndex] = currentSeat ? null : user; io.to(roomId).emit('update_seats', rooms[roomId].seats); } }); // লাইভ গিফট পাঠানো (Virtual Gift) socket.on('send_gift', ({ roomId, giftName, senderName }) => { io.to(roomId).emit('gift_received', { senderName, giftName, timestamp: new Date().toLocaleTimeString() }); }); // লাইভ চ্যাট মেসেজ socket.on('send_message', ({ roomId, message, sender }) => { const msgData = { sender, message }; if (rooms[roomId]) rooms[roomId].messages.push(msgData); io.to(roomId).emit('new_message', msgData); }); }); server.listen(3000, () => { console.log('MRChat Server running on port 3000'); });
